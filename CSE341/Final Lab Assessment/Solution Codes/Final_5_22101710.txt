Part A:




.MODEL SMALL
.STACK 100H


.DATA
    COURSES DB 'CSE110$', 'CSE111$', 'CSE220$', 'CSE221$', 'CSE341$', 'CSE340$', 'CSE471$', 'CSE422$', 'CSE423$', 'CSE460$'
    USER_INP DB 4 DUP(0)       
    MSG DB 'Enter 4-digit course indices (0-9): $'
    MSGADD DB 'Added courses:$'  
    NEW_LINE DB 0Dh, 0Ah, '$' 


.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX


    LEA DX, MSG
    MOV AH, 09H
    INT 21H


    MOV CX, 4
    LEA SI, USER_INP
READ_INPUT:
    MOV AH, 01H
    INT 21H
    SUB AL, '0'
    MOV [SI], AL
    INC SI
    LOOP READ_INPUT


    LEA DX, NEW_LINE
    MOV AH, 09H
    INT 21H


    LEA DX, MSGADD
    MOV AH, 09H
    INT 21H


    LEA DX, NEW_LINE
    MOV AH, 09H
    INT 21H


    MOV CX, 4
    LEA SI, USER_INP
PRINT_COURSES:
    MOV AL, [SI]        
    MOV BL, AL          
    MOV AX, BX          
    MOV BX, 7           
    MUL BX              
    ADD AX, OFFSET COURSES 
    MOV DX, AX          


    MOV AH, 09H
    INT 21H


    LEA DX, NEW_LINE
    MOV AH, 09H
    INT 21H


    INC SI              
    LOOP PRINT_COURSES


    MOV AX, 4CH
    INT 21H


MAIN ENDP
END MAIN










Part B:


.MODEL SMALL
.STACK 100H


.DATA
    COURSES DB 'CSE110', 'CSE111', 'CSE220', 'CSE221', 'CSE341', 'CSE340', 'CSE471', 'CSE422', 'CSE423', 'CSE460'
    USER_INP DB 4 DUP(?)   
    MSG DB 'ENTER UP TO 4 DIGIT COURSE INDEXES: $'
    MSGADD DB 'ADDED COURSES:', 0DH, 0AH, '$'
    SEMESTER DW 10000      
    TUITION DW 20000      
    TOTAL DW ?           


.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX


    LEA DX, MSG
    MOV AH, 09H
    INT 21H


    MOV SI, OFFSET USER_INP
    MOV CX, 4
INPUT_LOOP:
    MOV AH, 01H
    INT 21H
    CMP AL, 0DH    
    JE CALCULATE
    MOV [SI], AL   
    INC SI
    LOOP INPUT_LOOP


CALCULATE:
    MOV AX, SEMESTER
    MOV TOTAL, AX


    MOV SI, OFFSET USER_INP
    MOV CX, 4
    XOR BX, BX         


COURSE_LOOP:
    MOV AL, [SI]       
    CMP AL, 0          
    JE DONE_COURSE     
    SUB AL, '0'        
    INC BX             
    INC SI             
    LOOP COURSE_LOOP


DONE_COURSE:
    MOV AX, BX
    MOV DX, 0
    MOV CX, TUITION
    MUL CX             
    ADD TOTAL, AX      


ENDPGM:
    MOV AH, 4CH
    INT 21H


MAIN ENDP
END MAIN








Part C:


.MODEL SMALL
.STACK 100H


.DATA
    COURSES DB 'CSE110', 'CSE111', 'CSE220', 'CSE221', 'CSE341', 'CSE340', 'CSE471', 'CSE422', 'CSE423', 'CSE460'
    USER_INP DB ?
    MSG DB 'ENTER COURSE LEVEL (1, 2, 3, OR 4): $'
    MSG1 DB '100-LEVEL COURSES:', 0DH, 0AH, '$'
    MSG2 DB '200-LEVEL COURSES:', 0DH, 0AH, '$'
    MSG3 DB '300-LEVEL COURSES:', 0DH, 0AH, '$'
    MSG4 DB '400-LEVEL COURSES:', 0DH, 0AH, '$'
    COURSE_END DB 0DH, 0AH, '$'


.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX


    LEA DX, MSG
    MOV AH, 09H
    INT 21H


    MOV AH, 01H
    INT 21H
    SUB AL, '0'
    MOV USER_INP, AL
    
    CMP AL, 1
    JE LEVEL1
    CMP AL, 2
    JE LEVEL2
    CMP AL, 3
    JE LEVEL3
    CMP AL, 4
    JE LEVEL4
    JMP EXIT


LEVEL1:
    LEA DX, MSG1
    MOV AH, 09H
    INT 21H


    MOV SI, 0
    CALL PRINT_COURSE
    CALL PRINT_COURSE
    JMP EXIT


LEVEL2:
    LEA DX, MSG2
    MOV AH, 09H
    INT 21H


    MOV SI, 12
    CALL PRINT_COURSE
    CALL PRINT_COURSE
    JMP EXIT


LEVEL3:
    LEA DX, MSG3
    MOV AH, 09H
    INT 21H


    MOV SI, 24
    CALL PRINT_COURSE
    CALL PRINT_COURSE
    JMP EXIT


LEVEL4:
    LEA DX, MSG4
    MOV AH, 09H
    INT 21H


    MOV SI, 36
    CALL PRINT_COURSE
    CALL PRINT_COURSE
    CALL PRINT_COURSE
    CALL PRINT_COURSE


EXIT:
    MOV AH, 4CH
    INT 21H


PRINT_COURSE PROC
    MOV CX, 6
PRINT_LOOP:
    MOV AL, [SI]
    MOV DL, AL
    MOV AH, 02H
    INT 21H
    INC SI
    LOOP PRINT_LOOP
    MOV DX, OFFSET COURSE_END
    MOV AH, 09H
    INT 21H
    RET
PRINT_COURSE ENDP


END MAIN
